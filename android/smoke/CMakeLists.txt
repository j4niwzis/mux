cmake_minimum_required(VERSION 4.3.4)
include(../../cmake/import-std.cmake)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_PROJECT_TOP_LEVEL_INCLUDES "${CMAKE_CURRENT_LIST_DIR}/../../cmake/get_cme.cmake")
project(mux_android_smoke VERSION 0.1 LANGUAGES C CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "Configure with the Android toolchain")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(../../cmake/android.cmake)
cme_find_package(SDL3 3.5.0 REQUIRED)
add_library(mux SHARED main.cc ../../src/main_android.cc ../../src/android_runtime.cc)
target_sources(mux PRIVATE FILE_SET CXX_MODULES BASE_DIRS ../../src FILES
  ../../src/platform/files_android.cc ../../src/platform/window_setup_android.cc)
target_compile_options(mux PRIVATE
  $<$<AND:$<COMPILE_LANG_AND_ID:CXX,Clang>,$<VERSION_GREATER_EQUAL:$<CXX_COMPILER_VERSION>,23>>:-fno-experimental-new-constant-interpreter>)
target_link_libraries(mux PRIVATE SDL3::SDL3 GLESv2)
if(MANDK_ROOT)
  target_link_libraries(mux PRIVATE c++ c++abi unwind)
endif()
target_link_options(mux PRIVATE ${MUX_ANDROID_LINK_OPTIONS})
set(MUX_ANDROID_MANIFEST "${CMAKE_CURRENT_LIST_DIR}/AndroidManifest.xml")
include(../../cmake/android-apk.cmake)
mux_add_android_apk(mux)
