cmake_minimum_required(VERSION 3.24)
project(about_target_graph NONE)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/about_targets.cmake")
function(cme_port_field out port field)
  get_property(value GLOBAL PROPERTY TEST_${port}_${field})
  set(${out} "${value}" PARENT_SCOPE)
endfunction()
add_library(scene INTERFACE)
add_library(skiff::skiff ALIAS scene)
add_library(widgets INTERFACE)
add_library(skiff::widgets ALIAS widgets)
add_library(framework INTERFACE)
add_library(unused INTERFACE)
set_property(GLOBAL PROPERTY TEST_skiff_TARGETS skiff::skiff)
set_property(GLOBAL PROPERTY TEST_skiff-widgets_TARGETS skiff::widgets)
set_property(GLOBAL PROPERTY TEST_skiff-framework_TARGETS framework)
set_property(GLOBAL PROPERTY TEST_unused_TARGETS unused)
target_link_libraries(framework INTERFACE "$<BUILD_INTERFACE:skiff::skiff>" "$<LINK_ONLY:skiff::widgets>")
target_link_libraries(widgets INTERFACE skiff::skiff)
add_subdirectory(local-import)
mux_about_link_dependencies("skiff;skiff-widgets;skiff-framework;unused")
get_property(deps GLOBAL PROPERTY MUX_ABOUT_LINKS_skiff-framework)
if(NOT skiff IN_LIST deps OR NOT skiff-widgets IN_LIST deps OR unused IN_LIST deps)
  message(FATAL_ERROR "Already-created linked targets missing from dependencies: ${deps}")
endif()
get_property(deps GLOBAL PROPERTY MUX_ABOUT_LINKS_skiff-widgets)
if(NOT skiff IN_LIST deps)
  message(FATAL_ERROR "Transitive target edge missing: ${deps}")
endif()
